function(upgrade_test name)
  cmake_parse_arguments(TEST "" "TIMEOUT;FIXTURES_SETUP;FIXTURES_REQUIRED"
    "SOURCES;LIBRARIES;ARGUMENTS;ENVIRONMENT;LABELS" ${ARGN})
  if(NOT TEST_SOURCES)
    set(TEST_SOURCES ${name}.cpp)
  endif()
  if(NOT TEST_TIMEOUT)
    set(TEST_TIMEOUT 90)
  endif()
  add_executable(${name} ${TEST_SOURCES})
  target_link_libraries(${name} PRIVATE ${TEST_LIBRARIES})
  add_test(NAME ${name} COMMAND ${name} ${TEST_ARGUMENTS})
  set_tests_properties(${name} PROPERTIES TIMEOUT ${TEST_TIMEOUT})
  foreach(property ENVIRONMENT LABELS FIXTURES_SETUP FIXTURES_REQUIRED)
    if(TEST_${property})
      set_tests_properties(${name} PROPERTIES ${property} "${TEST_${property}}")
    endif()
  endforeach()
endfunction()
